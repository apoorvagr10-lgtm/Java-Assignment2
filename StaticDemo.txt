class StaticDemo {
    static int count = 0;

    StaticDemo() {
        count++;
    }

    static void showCount() {
        System.out.println("Total Objects: " + count);
    }

    public static void main(String[] args) {
        new StaticDemo();
        new StaticDemo();
        new StaticDemo();

        StaticDemo.showCount();
    }
}
